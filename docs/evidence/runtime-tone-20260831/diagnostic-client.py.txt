class ToneWebSocket(LoopbackWebSocket):
    def receive_json(self):
        while True:
            first, second = self._read_exact(2)
            if first not in (0x81, 0x89, 0x8a) or second & 0x80:
                raise ValueError('unexpected WebSocket frame')
            size = second & 127
            if first != 0x81 and size > 125:
                raise ValueError('oversized control frame')
            if size == 126:
                size = struct.unpack('!H', self._read_exact(2))[0]
            elif size == 127:
                size = struct.unpack('!Q', self._read_exact(8))[0]
            if size > self.endpoint.maximum_frame_bytes:
                raise ValueError('oversized frame')
            payload = self._read_exact(size)
            if first == 0x89:
                mask = os.urandom(4)
                masked = bytes(v ^ mask[i % 4] for i, v in enumerate(payload))
                self._socket().sendall(bytes((0x8a, 0x80 | size)) + mask + masked)
                continue
            if first == 0x8a:
                continue
            return json.loads(payload)

result = {'frequencyHz':14097100,'durationMs':10000,'gpio':20,'before':snapshot(),'messages':[]}
request = {'command':'bounded_tone','request_id':'runtime-tone-20260831-09',
    'frequency_source':'custom_rf','frequency_hz':14097100,'duration_ms':10000,
    'rp1_development':{'enabled':True,'route':'GPIO20','physical_connection':True,
        'attenuation_and_load':True,'bounded_operation':True,'non_radiating_topology':True,
        'experimental_acknowledged':True}}
try:
    with ToneWebSocket(BoundedToneEndpoint('127.0.0.1',31426,maximum_frame_bytes=1048576),time.monotonic()+30) as stream:
        stream.send_json(request)
        while True:
            reply = stream.receive_json()
            if reply.get('command') != 'bounded_tone': continue
            result['messages'].append(reply)
            print(json.dumps(reply),flush=True)
            if reply.get('event') == 'completed' or reply.get('started') is False: break
            if reply.get('started'):
                time.sleep(1)
                result['during'] = snapshot()
                print(json.dumps({'during':result['during']}),flush=True)
finally:
    result['after'] = snapshot()
    Path('/home/pi/runtime-tone-result.json').write_text(json.dumps(result,indent=2)+'\n')
    print(json.dumps({'after':result['after']}),flush=True)
